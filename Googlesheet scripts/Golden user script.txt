function l2i(letter) {
  var col = 0, len = letter.length;
  for (var i = 0; i < len; i++) {
    col += (letter.charCodeAt(i) - 64) * Math.pow(26, len - i - 1);
  }
  return col;
}

function norm(str) {
  return (str || "")
    .toString()
    .trim()
    .toUpperCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/\s+/g, "");
}

function normEmail(str) {
  return (str || "").toString().trim().toLowerCase();
}

function getPlatformType(p) {
  var val = (p || "").toString().trim().toUpperCase();

  if (val.indexOf("ZEFFY") !== -1 && val.indexOf("AEBEQ") !== -1) return "ZEFFY_AEBEQ";
  if (val.indexOf("ZEFFY") !== -1 && val.indexOf("SEMBEQ") !== -1) return "ZEFFY_SEMBEQ";
  if (val.indexOf("SALES") !== -1) return "SALESFORCE";
  if (val.indexOf("PRODON") !== -1) return "PRODON";
  if (val.indexOf("ZEFFY") !== -1) return "ZEFFY_OTHER";

  return "OTHER";
}

// ==================================================================
// LIMPIEZA - MOTOR
// Reglas:
// - Grupo estricto: Nombre + Apellido + CP
// - Grupo amplio: Nombre + Apellido
// - Verde: mismo Nombre + Apellido + CP, sin conflicto real
// - Amarillo: mismo Nombre + Apellido, pero CP distinto
// - Rojo: en el mismo Nombre + Apellido + CP, misma fuente con IDs distintos
// - Transparente: sin ID usable / sin email usable
// - En rojo NO copia nada
// - Zeffy usa EMAIL como ID, separado en AEBEQ y SEMBEQ
// ==================================================================
function procesarLimpiezaDesdePanel(cStr) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var n = data.length - 1;
  var lang = cStr.lang || 'FR';

  if (n < 1) {
    throw new Error(lang === 'FR' ? 'Pas assez de données.' : 'Not enough data.');
  }

  var cF  = l2i(cStr.c1);
  var cL  = l2i(cStr.c2);
  var cP  = l2i(cStr.c3);
  var cId = l2i(cStr.c4);
  var cEm = l2i(cStr.c5);
  var cPl = l2i(cStr.c6);

  var cCt = l2i(cStr.c7);
  var cSm = l2i(cStr.c8);
  var cDf = l2i(cStr.c9);
  var cSF = l2i(cStr.c10);
  var cPr = l2i(cStr.c11);
  var cZA = l2i(cStr.c12);
  var cZS = l2i(cStr.c13);

  var ctOut = [], smOut = [], dfOut = [], sfOut = [], prOut = [], zaOut = [], zsOut = [];
  for (var x = 0; x < n; x++) {
    ctOut.push([""]);
    smOut.push([""]);
    dfOut.push([""]);
    sfOut.push([""]);
    prOut.push([""]);
    zaOut.push([""]);
    zsOut.push([""]);
  }

  var mapStrict = {};
  var mapBroad = {};

  for (var i = 1; i <= n; i++) {
    var firstName = (data[i][cF - 1] || "").toString().trim();
    var lastName  = (data[i][cL - 1] || "").toString().trim();
    var cpRaw     = (data[i][cP - 1] || "").toString().trim().toUpperCase();
    var rawId     = (data[i][cId - 1] || "").toString().trim();
    var email     = normEmail(data[i][cEm - 1] || "");
    var platform  = (data[i][cPl - 1] || "").toString().trim();
    var pType     = getPlatformType(platform);

    if (!firstName && !lastName) continue;

    var keyBroad = norm(firstName) + "|" + norm(lastName);
    var keyStrict = norm(firstName) + "|" + norm(lastName) + "|" + norm(cpRaw);

    if (!mapBroad[keyBroad]) {
      mapBroad[keyBroad] = {
        rows: [],
        cps: new Set(),
        hasEmptyCP: false
      };
    }

    mapBroad[keyBroad].rows.push(i);
    if (cpRaw) mapBroad[keyBroad].cps.add(cpRaw);
    else mapBroad[keyBroad].hasEmptyCP = true;

    if (!mapStrict[keyStrict]) {
      mapStrict[keyStrict] = {
        rows: [],
        emails: new Set(),
        sfIds: new Set(),
        prIds: new Set(),
        zfAebeq: new Set(),
        zfSembeq: new Set()
      };
    }

    var gs = mapStrict[keyStrict];
    gs.rows.push(i);

    if (email) gs.emails.add(email);

    if (pType === "SALESFORCE") {
      if (rawId) gs.sfIds.add(rawId);
    } else if (pType === "PRODON") {
      if (rawId) gs.prIds.add(rawId);
    } else if (pType === "ZEFFY_AEBEQ") {
      if (email) gs.zfAebeq.add(email);
    } else if (pType === "ZEFFY_SEMBEQ") {
      if (email) gs.zfSembeq.add(email);
    }
  }

  function hasUsableData(groupStrict) {
    return (
      groupStrict.sfIds.size > 0 ||
      groupStrict.prIds.size > 0 ||
      groupStrict.zfAebeq.size > 0 ||
      groupStrict.zfSembeq.size > 0
    );
  }

  function isRedConflict(groupStrict) {
  return (
    groupStrict.sfIds.size > 1 ||
    groupStrict.prIds.size > 1
  );
}

  var redText = lang === 'FR' ? "🔴 Conflit ID" : "🔴 ID Conflict";
  var yellowText = lang === 'FR' ? "🟡 Vérifier CP" : "🟡 Check Postal Code";
  var greenText = lang === 'FR' ? "🟢 OK Match" : "🟢 OK Match";

  for (var kb in mapBroad) {
    var broadGroup = mapBroad[kb];
    var isYellowGroup = broadGroup.rows.length > 1 && (broadGroup.cps.size > 1 || broadGroup.hasEmptyCP);

    for (var r = 0; r < broadGroup.rows.length; r++) {
      var rowIdx = broadGroup.rows[r];

      var firstName = (data[rowIdx][cF - 1] || "").toString().trim();
      var lastName  = (data[rowIdx][cL - 1] || "").toString().trim();
      var cpRaw     = (data[rowIdx][cP - 1] || "").toString().trim().toUpperCase();

      var keyStrict = norm(firstName) + "|" + norm(lastName) + "|" + norm(cpRaw);
      var strictGroup = mapStrict[keyStrict];

      var red = isRedConflict(strictGroup);
      var transparent = !hasUsableData(strictGroup);

      ctOut[rowIdx - 1] = [broadGroup.rows.length];
      dfOut[rowIdx - 1] = [strictGroup.emails.size];

      if (red) {
        smOut[rowIdx - 1] = [redText];
        sfOut[rowIdx - 1] = [""];
        prOut[rowIdx - 1] = [""];
        zaOut[rowIdx - 1] = [""];
        zsOut[rowIdx - 1] = [""];
        continue;
      }

      if (transparent) {
        smOut[rowIdx - 1] = [""];
        sfOut[rowIdx - 1] = [""];
        prOut[rowIdx - 1] = [""];
        zaOut[rowIdx - 1] = [""];
        zsOut[rowIdx - 1] = [""];
        continue;
      }

      smOut[rowIdx - 1] = [isYellowGroup ? yellowText : greenText];

      sfOut[rowIdx - 1] = [strictGroup.sfIds.size === 1 ? Array.from(strictGroup.sfIds)[0] : ""];
      prOut[rowIdx - 1] = [strictGroup.prIds.size === 1 ? Array.from(strictGroup.prIds)[0] : ""];
      zaOut[rowIdx - 1] = [strictGroup.zfAebeq.size === 1 ? Array.from(strictGroup.zfAebeq)[0] : ""];
      zsOut[rowIdx - 1] = [strictGroup.zfSembeq.size === 1 ? Array.from(strictGroup.zfSembeq)[0] : ""];
    }
  }

  sheet.getRange(2, cCt, n, 1).setValues(ctOut);
  sheet.getRange(2, cSm, n, 1).setValues(smOut);
  sheet.getRange(2, cDf, n, 1).setValues(dfOut);
  sheet.getRange(2, cSF, n, 1).setValues(sfOut);
  sheet.getRange(2, cPr, n, 1).setValues(prOut);
  sheet.getRange(2, cZA, n, 1).setValues(zaOut);
  sheet.getRange(2, cZS, n, 1).setValues(zsOut);

  return lang === 'FR' ? 'Nettoyage terminé.' : 'Cleaning completed.';
}

function procesarMergeDesdePanel(cStr) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data  = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) throw new Error("Pas assez de données.");

  function l2iMerge(letter) {
    var col = 0, len = letter.length;
    for (var i = 0; i < len; i++) {
      col += (letter.charCodeAt(i) - 64) * Math.pow(26, len - i - 1);
    }
    return col - 1;

    return (semaforo || "").toString().indexOf("🔴") !== -1;

    var p = (platRaw || "").toString().toUpperCase();
    return p.indexOf("ZEFFY") !== -1 && p.indexOf("SEMBEQ") !== -1;
  }

  function isSalesforce(platRaw) {
    return (platRaw || "").toString().toUpperCase().indexOf("SALES") !== -1;
  }

  function isProdon(platRaw) {
    return (platRaw || "").toString().toUpperCase().indexOf("PRODON") !== -1;

  function countFilledParts(ad, cp, ville, province, pays) {
    var score = 0;
    if ((ad || "").toString().trim()) score++;
    if ((cp || "").toString().trim()) score++;
    if ((ville || "").toString().trim()) score++;
    if ((province || "").toString().trim()) score++;
    if ((pays || "").toString().trim()) score++;
    return score;
  }

  function nbValueRank(v) {
    var t = (v || "").toString().trim().toUpperCase();
    if (t === "+4") return 5;
    var n = parseInt(t, 10);
    return isNaN(n) ? 0 : n;
  }

  function bestNbValue(currentVal, newVal) {
    return nbValueRank(newVal) > nbValueRank(currentVal) ? newVal : currentVal;
  }

function setToSortedString(s, sep) {
  return Array.from(s).sort().join(sep || " | ");
}

function getEmailKeysSFProdon(emailMap) {
  var salesforceKey = "";
  var prodonKey = "";

  for (var k in emailMap) {
    var up = (k || "").toString().toUpperCase();
    if (!salesforceKey && up.indexOf("SALES") !== -1) salesforceKey = k;
    if (!prodonKey && up.indexOf("PRODON") !== -1) prodonKey = k;
  }

  return {
    salesforceKey: salesforceKey,
    prodonKey: prodonKey
  };
}

function getEmailsSalesforceOutput(emailMap) {
  var keys = getEmailKeysSFProdon(emailMap);
  return keys.salesforceKey && emailMap[keys.salesforceKey]
    ? Array.from(emailMap[keys.salesforceKey]).sort()
    : [];
}

function getEmailsOtherOutput(emailMap) {
  var keys = getEmailKeysSFProdon(emailMap);
  var sfSet = keys.salesforceKey && emailMap[keys.salesforceKey]
    ? emailMap[keys.salesforceKey]
    : new Set();

  var prodonArr = keys.prodonKey && emailMap[keys.prodonKey]
    ? Array.from(emailMap[keys.prodonKey]).sort()
    : [];

  return prodonArr.filter(function(email) {
    return !sfSet.has(email);
  });
}

function normAdresse(str) {
  return (str || "")
    .toString()
    .trim()
    .toUpperCase()
    .normalize("NFD").replace(/[\u0300-\u036f]/g, "")
    .replace(/[^A-Z0-9\s]/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

function buildAdresseLocale(adresse, cp, ville, province, pays) {
  return [adresse, ville, province, cp, pays].filter(Boolean).join(", ");
}

function buildAdresseKeyForCache(adresse, cp, ville, province, pays) {
  return [
    normAdresse(adresse),
    normAdresse(cp),
    normAdresse(ville),
    normAdresse(province),
    normAdresse(pays)
  ].join("|");
}

function extractGoogleAddressParts(result) {
  var comps = result.address_components || [];

  function getComp(type) {
    for (var i = 0; i < comps.length; i++) {
      if (comps[i].types && comps[i].types.indexOf(type) !== -1) {
        return comps[i].long_name || "";
      }
    }
    return "";
  }

  var streetNumber = getComp("street_number");
  var route        = getComp("route");
  var subpremise   = getComp("subpremise");
  var locality     = getComp("locality") || getComp("postal_town") || getComp("sublocality") || getComp("administrative_area_level_3");
  var province     = getComp("administrative_area_level_1");
  var country      = getComp("country");
  var postalCode   = getComp("postal_code");

  var street = [streetNumber, route].filter(Boolean).join(" ").trim();
  if (subpremise) street = street ? street + ", " + subpremise : subpremise;

  return {
    adresseNorm: buildAdresseKeyForCache(street, postalCode, locality, province, country),
    adresseOut: street || (result.formatted_address || ""),
    cpOut: postalCode || "",
    villeOut: locality || "",
    provinceOut: province || "",
    paysOut: country || ""
  };
}

function geocodeAdresseGoogle(fullAddress, apiKey) {
  if (!apiKey) throw new Error("API key Google manquante.");

  var url = "https://maps.googleapis.com/maps/api/geocode/json"
    + "?address=" + encodeURIComponent(fullAddress)
    + "&language=fr"
    + "&region=ca"
    + "&key=" + encodeURIComponent(apiKey);

  var response = UrlFetchApp.fetch(url, {
    muteHttpExceptions: true,
    method: "get",
    contentType: "application/json"
  });

  var code = response.getResponseCode();
  if (code !== 200) {
    throw new Error("Google HTTP " + code);
  }

  var json = JSON.parse(response.getContentText());
  if (json.status !== "OK" || !json.results || !json.results.length) {
    throw new Error("Google " + (json.status || "UNKNOWN"));
  }

  return extractGoogleAddressParts(json.results[0]);
}

function canadaPostFind(fullAddress, apiKey) {
  if (!apiKey) throw new Error("API key Canada Post manquante.");

  var url = "https://ws1.postescanada-canadapost.ca/addresscomplete/interactive/find/v2.10/json3.ws"
    + "?Key=" + encodeURIComponent(apiKey)
    + "&SearchTerm=" + encodeURIComponent(fullAddress)
    + "&Country=CAN"
    + "&LanguagePreference=FR"
    + "&MaxSuggestions=7";

  var response = UrlFetchApp.fetch(url, {
    muteHttpExceptions: true,
    method: "get",
    contentType: "application/json"
  });

  var code = response.getResponseCode();
  if (code !== 200) {
    throw new Error("Canada Find HTTP " + code);
  }

  var json = JSON.parse(response.getContentText());
  if (!json || !json.Items || !json.Items.length) {
    throw new Error("Canada Find EMPTY");
  }

  if (json.Items[0] && json.Items[0].Error) {
    throw new Error("Canada Find " + json.Items[0].Error + ": " + json.Items[0].Description);
  }

  for (var i = 0; i < json.Items.length; i++) {
    if ((json.Items[i].Next || "").toString().toUpperCase() === "RETRIEVE") {
      return json.Items[i];
    }
  }

  return json.Items[0];
}

function canadaPostRetrieve(itemId, apiKey) {
  var url = "https://ws1.postescanada-canadapost.ca/addresscomplete/interactive/retrieve/v2.11/json3.ws"
    + "?Key=" + encodeURIComponent(apiKey)
    + "&Id=" + encodeURIComponent(itemId);

  var response = UrlFetchApp.fetch(url, {
    muteHttpExceptions: true,
    method: "get",
    contentType: "application/json"
  });

  var code = response.getResponseCode();
  if (code !== 200) {
    throw new Error("Canada Retrieve HTTP " + code);
  }

  var json = JSON.parse(response.getContentText());
  if (!json || !json.Items || !json.Items.length) {
    throw new Error("Canada Retrieve EMPTY");
  }

  if (json.Items[0] && json.Items[0].Error) {
    throw new Error("Canada Retrieve " + json.Items[0].Error + ": " + json.Items[0].Description);
  }

  var item = json.Items[0];

  var adresse = [item.Line1, item.Line2].filter(Boolean).join(", ").trim();
  var cp = item.PostalCode || "";
  var ville = item.City || "";
  var province = item.ProvinceCode || item.ProvinceName || item.AdminAreaCode || item.AdminAreaName || "";
  var pays = item.CountryIso2 || item.CountryName || "";

  return {
    adresseNorm: buildAdresseKeyForCache(adresse, cp, ville, province, pays),
    adresseOut: adresse,
    cpOut: cp,
    villeOut: ville,
    provinceOut: province,
    paysOut: pays
  };
}

function geocodeAdresseCanadaPost(fullAddress, apiKey) {
  var found = canadaPostFind(fullAddress, apiKey);
  if (!found || !found.Id) {
    throw new Error("Canada Find sans Id");
  }
  return canadaPostRetrieve(found.Id, apiKey);
}

function procesarAdresseDesdePanel(cStr) {
  var provider = (cStr.c13 || "").toString().trim().toUpperCase();
  var apiKey = (cStr.c14 || "").toString().trim();
  var statusToProcess = (cStr.c15 || "").toString().trim().toUpperCase();
  var maxRun = parseInt(cStr.c16, 10);

  if (!statusToProcess) {
    return "Status vide: rien traité.";
  }

  if (provider !== "GOOGLE" && provider !== "CANADA") {
    throw new Error("Provider invalide. Choisis GOOGLE ou CANADA.");
  }

  if (!apiKey) {
    throw new Error("API Key manquante.");
  }

  if (isNaN(maxRun) || maxRun <= 0) {
    throw new Error("Max rows this run invalide.");
  }

  var props = PropertiesService.getDocumentProperties();
  var current = getAdresseJobState_();

  var newConfig = {
    c1: cStr.c1, c2: cStr.c2, c3: cStr.c3, c4: cStr.c4, c5: cStr.c5,
    c6: cStr.c6, c7: cStr.c7, c8: cStr.c8, c9: cStr.c9, c10: cStr.c10,
    c11: cStr.c11, c12: cStr.c12, c13: provider, c14: apiKey, c15: statusToProcess, c16: String(maxRun)
  };

  var newSignature = JSON.stringify({
    c1: newConfig.c1, c2: newConfig.c2, c3: newConfig.c3, c4: newConfig.c4, c5: newConfig.c5,
    c6: newConfig.c6, c7: newConfig.c7, c8: newConfig.c8, c9: newConfig.c9, c10: newConfig.c10,
    c11: newConfig.c11, c12: newConfig.c12, c13: newConfig.c13, c15: newConfig.c15, c16: newConfig.c16
  });

  var oldSignature = props.getProperty('ADRESSE_JOB_SIGNATURE') || '';

  var shouldRestart = !current.mode || current.mode === 'COMPLETED' || current.mode === 'IDLE' || oldSignature !== newSignature;

  if (shouldRestart) {
    clearAdresseTrigger_();
    props.setProperties({
      ADRESSE_JOB_CONFIG: JSON.stringify(newConfig),
      ADRESSE_JOB_SIGNATURE: newSignature,
      ADRESSE_JOB_MODE: 'RUNNING',
      ADRESSE_JOB_LAST_ROW: '1',
      ADRESSE_JOB_PROCESSED_COUNT: '0',
      ADRESSE_JOB_MESSAGE: 'Démarrage nouveau job'
    }, true);
  } else {
    props.setProperty('ADRESSE_JOB_MODE', 'RUNNING');
    props.setProperty('ADRESSE_JOB_MESSAGE', 'Reprise du job');
    props.setProperty('ADRESSE_JOB_CONFIG', JSON.stringify(newConfig));
  }

  var res = runAdresseJobChunk_();
  return res;
}

function getAdresseJobState_() {
  var props = PropertiesService.getDocumentProperties();
  var configRaw = props.getProperty('ADRESSE_JOB_CONFIG');
  var config = configRaw ? JSON.parse(configRaw) : null;

  return {
    mode: props.getProperty('ADRESSE_JOB_MODE') || 'IDLE',
    provider: config ? config.c13 : '',
    statusToProcess: config ? config.c15 : '',
    maxTargetRows: parseInt(config ? config.c16 : '0', 10) || 0,
    processedCount: parseInt(props.getProperty('ADRESSE_JOB_PROCESSED_COUNT') || '0', 10),
    lastScanRow: parseInt(props.getProperty('ADRESSE_JOB_LAST_ROW') || '1', 10),
    message: props.getProperty('ADRESSE_JOB_MESSAGE') || '',
    triggerActive: hasAdresseTrigger_()
  };
}

function getAdresseJobState() {
  return getAdresseJobState_();
}

function hasAdresseTrigger_() {
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === 'adresseJobTriggerRunner') {
      return true;
    }
  }
  return false;
}

function clearAdresseTrigger_() {
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === 'adresseJobTriggerRunner') {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }
}

function scheduleNextAdresseRun_() {
  clearAdresseTrigger_();
  ScriptApp.newTrigger('adresseJobTriggerRunner')
    .timeBased()
    .after(60 * 1000)
    .create();
}

function pauseAdresseJob() {
  var props = PropertiesService.getDocumentProperties();
  props.setProperty('ADRESSE_JOB_MODE', 'PAUSED');
  props.setProperty('ADRESSE_JOB_MESSAGE', 'Job en pause');
  clearAdresseTrigger_();
  return 'Job en pause. Progression conservée.';
}

function stopAdresseJob() {
  var props = PropertiesService.getDocumentProperties();
  props.setProperty('ADRESSE_JOB_MODE', 'STOPPED');
  props.setProperty('ADRESSE_JOB_MESSAGE', 'Job arrêté');
  clearAdresseTrigger_();
  return 'Job arrêté. Progression conservée.';
}

function adresseJobTriggerRunner() {
  runAdresseJobChunk_();
}

function runAdresseJobChunk_() {
  var lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    var props = PropertiesService.getDocumentProperties();
    var mode = props.getProperty('ADRESSE_JOB_MODE') || 'IDLE';

    if (mode !== 'RUNNING') {
      clearAdresseTrigger_();
      return 'Job non actif.';
    }

    var configRaw = props.getProperty('ADRESSE_JOB_CONFIG');
    if (!configRaw) {
      clearAdresseTrigger_();
      props.setProperty('ADRESSE_JOB_MODE', 'IDLE');
      props.setProperty('ADRESSE_JOB_MESSAGE', 'Configuration absente');
      return 'Configuration absente.';
    }

    var cStr = JSON.parse(configRaw);

    var provider = (cStr.c13 || '').toString().trim().toUpperCase();
    var apiKey = (cStr.c14 || '').toString().trim();
    var statusToProcess = (cStr.c15 || '').toString().trim().toUpperCase();
    var maxTargetRows = parseInt(cStr.c16, 10);

    if (!statusToProcess) {
      props.setProperty('ADRESSE_JOB_MODE', 'STOPPED');
      props.setProperty('ADRESSE_JOB_MESSAGE', 'Status vide');
      clearAdresseTrigger_();
      return 'Status vide.';
    }

    if (!apiKey) {
      props.setProperty('ADRESSE_JOB_MODE', 'STOPPED');
      props.setProperty('ADRESSE_JOB_MESSAGE', 'API Key manquante');
      clearAdresseTrigger_();
      return 'API Key manquante.';
    }

    if (isNaN(maxTargetRows) || maxTargetRows <= 0) {
      props.setProperty('ADRESSE_JOB_MODE', 'STOPPED');
      props.setProperty('ADRESSE_JOB_MESSAGE', 'Max rows invalide');
      clearAdresseTrigger_();
      return 'Max rows invalide.';
    }

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getValues();
    var n = data.length - 1;

    if (n < 1) {
      props.setProperty('ADRESSE_JOB_MODE', 'COMPLETED');
      props.setProperty('ADRESSE_JOB_MESSAGE', 'Pas assez de données');
      clearAdresseTrigger_();
      return 'Pas assez de données.';
    }

    var cAdr  = l2i(cStr.c1);
    var cCP   = l2i(cStr.c2);
    var cPay  = l2i(cStr.c3);
    var cVil  = l2i(cStr.c4);
    var cProv = l2i(cStr.c5);

    var cAdrNorm = l2i(cStr.c6);
    var cAdrOut  = l2i(cStr.c7);
    var cCPOut   = l2i(cStr.c8);
    var cVilOut  = l2i(cStr.c9);
    var cProvOut = l2i(cStr.c10);
    var cPayOut  = l2i(cStr.c11);
    var cProc    = l2i(cStr.c12);

    var adrNormOut = [], adrOut = [], cpOut = [], vilOut = [], provOut = [], payOut = [], procOut = [];
    for (var x = 0; x < n; x++) {
      adrNormOut.push([data[x + 1][cAdrNorm - 1] || ""]);
      adrOut.push([data[x + 1][cAdrOut - 1] || ""]);
      cpOut.push([data[x + 1][cCPOut - 1] || ""]);
      vilOut.push([data[x + 1][cVilOut - 1] || ""]);
      provOut.push([data[x + 1][cProvOut - 1] || ""]);
      payOut.push([data[x + 1][cPayOut - 1] || ""]);
      procOut.push([data[x + 1][cProc - 1] || ""]);
    }

    var processedCount = parseInt(props.getProperty('ADRESSE_JOB_PROCESSED_COUNT') || '0', 10);
    var lastScanRow = parseInt(props.getProperty('ADRESSE_JOB_LAST_ROW') || '1', 10);

    if (processedCount >= maxTargetRows) {
      props.setProperty('ADRESSE_JOB_MODE', 'COMPLETED');
      props.setProperty('ADRESSE_JOB_MESSAGE', 'Objectif atteint');
      clearAdresseTrigger_();
      return 'Objectif déjà atteint.';
    }

    var processedNow = 0;
    var cache = {};
    var doneStatus = provider === "GOOGLE" ? "DONE_GOOGLE" : "DONE_CANADA";
    var errorStatus = provider === "GOOGLE" ? "ERROR_GOOGLE" : "ERROR_CANADA";
    var startMs = new Date().getTime();
    var maxChunkMs = 5.5 * 60 * 1000;
    var reachedEnd = true;
    var currentRowScanned = lastScanRow;

    for (var i = lastScanRow + 1; i <= n; i++) {
      currentRowScanned = i;

      if ((new Date().getTime() - startMs) > maxChunkMs) {
        reachedEnd = false;
        break;
      }

      if ((processedCount + processedNow) >= maxTargetRows) {
        reachedEnd = false;
        break;
      }

      var currentStatus = (data[i][cProc - 1] || "").toString().trim().toUpperCase();
      if (currentStatus !== statusToProcess) {
        continue;
      }

      var idx = i - 1;
      var adr  = (data[i][cAdr  - 1] || "").toString().trim();
      var cp   = (data[i][cCP   - 1] || "").toString().trim();
      var vil  = (data[i][cVil  - 1] || "").toString().trim();
      var prov = (data[i][cProv - 1] || "").toString().trim();
      var pay  = (data[i][cPay  - 1] || "").toString().trim();

      if (!adr && !cp && !vil) {
        procOut[idx] = ["SKIPPED"];
        processedNow++;
        continue;
      }

      var fullAddress = buildAdresseLocale(adr, cp, vil, prov, pay);
      var cacheKey = provider + "|" + buildAdresseKeyForCache(adr, cp, vil, prov, pay);

      try {
        var result;

        if (cache[cacheKey]) {
          result = cache[cacheKey];
        } else {
          if (provider === "GOOGLE") {
            result = geocodeAdresseGoogle(fullAddress, apiKey);
          } else {
            result = geocodeAdresseCanadaPost(fullAddress, apiKey);
          }
          cache[cacheKey] = result;
        }

        adrNormOut[idx] = [result.adresseNorm || ""];
        adrOut[idx]     = [result.adresseOut || ""];
        cpOut[idx]      = [result.cpOut || ""];
        vilOut[idx]     = [result.villeOut || ""];
        provOut[idx]    = [result.provinceOut || ""];
        payOut[idx]     = [result.paysOut || ""];
        procOut[idx]    = [doneStatus];
        processedNow++;

      } catch (e) {
        procOut[idx] = [errorStatus];
        processedNow++;
      }
    }

    sheet.getRange(2, cAdrNorm, n, 1).setValues(adrNormOut);
    sheet.getRange(2, cAdrOut,  n, 1).setValues(adrOut);
    sheet.getRange(2, cCPOut,   n, 1).setValues(cpOut);
    sheet.getRange(2, cVilOut,  n, 1).setValues(vilOut);
    sheet.getRange(2, cProvOut, n, 1).setValues(provOut);
    sheet.getRange(2, cPayOut,  n, 1).setValues(payOut);
    sheet.getRange(2, cProc,    n, 1).setValues(procOut);

    var newProcessedCount = processedCount + processedNow;
    props.setProperty('ADRESSE_JOB_PROCESSED_COUNT', String(newProcessedCount));
    props.setProperty('ADRESSE_JOB_LAST_ROW', String(currentRowScanned));

    var foundMoreMatchingAhead = false;
    if (!reachedEnd && currentRowScanned < n && newProcessedCount < maxTargetRows) {
      foundMoreMatchingAhead = true;
    }

    if (reachedEnd || newProcessedCount >= maxTargetRows) {
      props.setProperty('ADRESSE_JOB_MODE', 'COMPLETED');
      props.setProperty('ADRESSE_JOB_MESSAGE', 'Terminé');
      clearAdresseTrigger_();
    } else {
      props.setProperty('ADRESSE_JOB_MODE', 'RUNNING');
      props.setProperty('ADRESSE_JOB_MESSAGE', 'En attente de reprise automatique');
      scheduleNextAdresseRun_();
    }

    return "Provider: " + provider +
           " | Status traité: " + statusToProcess +
           " | Traité maintenant: " + processedNow +
           " | Total traité: " + newProcessedCount + "/" + maxTargetRows;
  } finally {
    lock.releaseLock();
  }
}

  var props = PropertiesService.getDocumentProperties();
  var mode = props.getProperty('ADRESSE_JOB_MODE') || 'IDLE';

function hasGreenSemaforo_(val) {
  return (val || '').toString().indexOf('🟢') !== -1;
}

function hasYellowSemaforo_(val) {
  return (val || '').toString().indexOf('🟡') !== -1;
}

function hasRedSemaforo_(val) {
  return (val || '').toString().indexOf('🔴') !== -1;
}

function normalizeCourrierText_(str) {
  return (str || '')
    .toString()
    .trim()
    .toUpperCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/\s+/g, ' ')
    .trim();
}

function normalizeCourrierNameKey_(firstName, lastName) {
  return normalizeCourrierText_(firstName) + '|' + normalizeCourrierText_(lastName);
}

function bucketNbPersonnes_(count) {
  count = parseInt(count, 10);
  if (isNaN(count) || count < 1) return '';
  return String(count);
}

function buildCourrierGroupId_(adresseNorm) {
  var raw = normalizeCourrierText_(adresseNorm);
  var digest = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, raw);
  var hex = digest.map(function(b) {
    var v = (b < 0 ? b + 256 : b).toString(16).toUpperCase();
    return v.length === 1 ? '0' + v : v;
  }).join('');
  return 'CG_' + hex.substring(0, 10);
}

function chooseCourrierRecipientPerson_(persons) {
  if (!persons || !persons.length) return null;

  var ordered = persons.slice().sort(function(a, b) {
    return a.firstRow - b.firstRow;
  });

  var males = ordered.filter(function(p) { return p.genre === 'M'; });
  if (males.length) return males[0];

  var females = ordered.filter(function(p) { return p.genre === 'F'; });
  if (females.length) return females[0];

  return ordered[0];
}

function procesarCourrierFoyerDesdePanel(cStr) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var n = data.length - 1;

  if (n < 1) {
    throw new Error('Pas assez de données.');
  }

  var cSem   = l2i(cStr.c1);
  var cMerge = l2i(cStr.c2);
  var cFirst = l2i(cStr.c3);
  var cLast  = l2i(cStr.c4);
  var cAddr  = l2i(cStr.c5);
  var cStat  = l2i(cStr.c6);
  var cGenre = l2i(cStr.c7);

  var cGrp   = l2i(cStr.c8);
  var cNb    = l2i(cStr.c9);
  var cRecv  = l2i(cStr.c10);

  var grpOut = [], nbOut = [], recvOut = [];
  for (var x = 0; x < n; x++) {
    grpOut.push([""]);
    nbOut.push([""]);
    recvOut.push([""]);
  }

  var addressMap = {};
  var validRows = 0;
  var ignoredRows = 0;

  for (var i = 1; i <= n; i++) {
    var idx = i - 1;

    var sem = (data[i][cSem - 1] || '').toString().trim();
    var mergeKey = (data[i][cMerge - 1] || '').toString().trim();
    var firstName = (data[i][cFirst - 1] || '').toString().trim();
    var lastName = (data[i][cLast - 1] || '').toString().trim();
    var addrNorm = (data[i][cAddr - 1] || '').toString().trim();
    var addrStatus = (data[i][cStat - 1] || '').toString().trim().toUpperCase();
    var genre = (data[i][cGenre - 1] || '').toString().trim().toUpperCase();

    if (!addrNorm) {
      ignoredRows++;
      continue;
    }

    if (addrStatus !== 'DONE_GOOGLE' && addrStatus !== 'DONE_CANADA') {
      ignoredRows++;
      continue;
    }

    if (hasRedSemaforo_(sem) || !sem) {
      ignoredRows++;
      continue;
    }

    var personKey = '';
    if (hasYellowSemaforo_(sem)) {
      if (!mergeKey) {
        ignoredRows++;
        continue;
      }
      personKey = 'MANUAL|' + normalizeCourrierText_(mergeKey);
    } else if (hasGreenSemaforo_(sem)) {
      personKey = 'NAME|' + normalizeCourrierNameKey_(firstName, lastName);
      if (personKey === 'NAME||') {
        ignoredRows++;
        continue;
      }
    } else {
      ignoredRows++;
      continue;
    }

    var addrKey = normalizeCourrierText_(addrNorm);
    if (!addressMap[addrKey]) {
      addressMap[addrKey] = {
        address: addrNorm,
        rows: [],
        persons: {}
      };
    }

    addressMap[addrKey].rows.push(i);

    if (!addressMap[addrKey].persons[personKey]) {
      addressMap[addrKey].persons[personKey] = {
        personKey: personKey,
        firstRow: i,
        firstName: firstName,
        lastName: lastName,
        genre: (genre === 'M' || genre === 'F') ? genre : '',
        rows: []
      };
    }

    var personRef = addressMap[addrKey].persons[personKey];
    personRef.rows.push(i);

    if (!personRef.genre && (genre === 'M' || genre === 'F')) {
      personRef.genre = genre;
    }

    validRows++;
  }

  var domicileCount = 0;

  for (var addrKey in addressMap) {
    domicileCount++;

    var group = addressMap[addrKey];
    var groupId = buildCourrierGroupId_(group.address);

    var persons = Object.keys(group.persons).map(function(pk) {
      return group.persons[pk];
    }).sort(function(a, b) {
      return a.firstRow - b.firstRow;
    });

    var nbPersons = persons.length;
    var bucket = bucketNbPersonnes_(nbPersons);
    var recipient = chooseCourrierRecipientPerson_(persons);
    var recipientRow = recipient ? recipient.firstRow : null;

    for (var p = 0; p < persons.length; p++) {
      var person = persons[p];
      for (var r = 0; r < person.rows.length; r++) {
        var sheetRow = person.rows[r];
        var outIdx = sheetRow - 1;

        grpOut[outIdx] = [groupId];
        nbOut[outIdx] = [bucket];
        recvOut[outIdx] = [sheetRow === recipientRow ? '1' : ''];
      }
    }
  }

  sheet.getRange(2, cGrp, n, 1).setValues(grpOut);
  sheet.getRange(2, cNb, n, 1).setValues(nbOut);
  sheet.getRange(2, cRecv, n, 1).setValues(recvOut);

  return 'Courrier/Foyer terminé. Domiciles: ' + domicileCount +
         ' | Lignes valides: ' + validRows +
         ' | Lignes ignorées: ' + ignoredRows;
}

    var mergeManual = (data[i][cMerge - 1] || '').toString().trim();
    var companyName = (data[i][cName - 1] || '').toString().trim();
    var addrNorm = (data[i][cAddr - 1] || '').toString().trim();

    var parsedAddr = companyParseAdresseNormalisee_(addrNorm);
    var tokens = companyStrongNameTokens_(companyName, parsedAddr.city);

    var rec = {
      row: i + 1,
      idx: idx,
      platform: platform,
      rawId: rawId,
      mergeManual: mergeManual,
      companyName: companyName,
      addrNorm: addrNorm,
      streetPart: parsedAddr.streetPart,
      postal: parsedAddr.postal,
      city: parsedAddr.city,
      province: parsedAddr.province,
      country: parsedAddr.country,
      postalKey: companyNormalizePostal_(parsedAddr.postal),
      cityKey: companyNormalizeText_(parsedAddr.city),
      addressBaseKey: parsedAddr.addressBaseKey,
      addressGroupKey: parsedAddr.addressGroupKey,
      addressBaseDisplay: parsedAddr.addressBaseDisplay,
      nameTokens: tokens,
      nameClean: tokens.join(' '),
      marked: false
    };

    records.push(rec);

    if (rec.addressGroupKey) {
      if (!groupsByAddress[rec.addressGroupKey]) groupsByAddress[rec.addressGroupKey] = [];
      groupsByAddress[rec.addressGroupKey].push(rec);
    }

    for (var t = 0; t < tokens.length; t++) {
      var tok = tokens[t];
      if (!groupsByNameToken[tok]) groupsByNameToken[tok] = [];
      groupsByNameToken[tok].push(rec);
    }
  }

  var groupCounter = 1;
  var statsStrong = 0;
  var statsReview = 0;
  var statsOrange = 0;
  var statsConflict = 0;

  for (var addrKey in groupsByAddress) {
    var group = groupsByAddress[addrKey];
    if (group.length < 2) continue;

    var groupCode = 'CMP-' + companyPad4_(groupCounter++);
    var postalSet = companyUniqueNonBlank_(group.map(function(r) { return r.postalKey; }));
    var platformSet = companyUniqueNonBlank_(group.map(function(r) { return companyNormalizeText_(r.platform); }));
    var idConflict = companyHasSameSourceDifferentIds_(group);
    var nameOverlap = companyGroupHasNameOverlap_(group);

    var color = '#fff2cc';
    var reasonPrefix = '🟡 SAME_BASE_REVIEW';
    var reason = '';

    if (idConflict) {
      color = '#f4cccc';
      reasonPrefix = '🔴 POSSIBLE_ID_CONFLICT';
      reason = 'Même adresse base, mais même plateforme avec No client différents.';
      statsConflict++;
    } else if (postalSet.length > 1 && nameOverlap) {
      color = '#fff2cc';
      reasonPrefix = '🟡 SAME_BASE_CP_DIFF';
      reason = 'Même adresse base + ville + nom compatible, mais CP différent: ' + postalSet.join(', ');
      statsReview++;
    } else if (postalSet.length > 1 && !nameOverlap) {
      color = '#f4cccc';
      reasonPrefix = '🔴 SAME_BASE_CP_AND_NAME_DIFF';
      reason = 'Même adresse base + ville, mais CP différent et nom pas clairement compatible: ' + postalSet.join(', ');
      statsConflict++;
    } else if (postalSet.length <= 1 && nameOverlap) {
      color = '#d9ead3';
      reasonPrefix = '🟢 SAME_ADDRESS_STRONG';
      reason = 'Même adresse base + ville + CP + nom compatible.';
      statsStrong++;
    } else {
      color = '#fff2cc';
      reasonPrefix = '🟡 SAME_ADDRESS_NAME_REVIEW';
      reason = 'Même adresse base + ville + CP, mais nom pas clairement compatible.';
      statsReview++;
    }

    if (platformSet.length > 1) {
      reason += ' | Plateformes: ' + platformSet.join(', ');
    }

    group.forEach(function(r) {
      markOut[r.idx] = [groupCode];
      reasonOut[r.idx] = [reasonPrefix + ' | ' + reason + ' | Base: ' + r.addressBaseDisplay];
      bgMark[r.idx] = [color];
      bgReason[r.idx] = [color];
      r.marked = true;
    });
  }

  var nameGroupCounter = 1;

  for (var token in groupsByNameToken) {
    var nameGroup = groupsByNameToken[token].filter(function(r) {
      return !r.marked && r.addressGroupKey;
    });

    if (nameGroup.length < 2) continue;

    var distinctAddresses = companyUniqueNonBlank_(nameGroup.map(function(r) { return r.addressGroupKey; }));
    if (distinctAddresses.length < 2) continue;

    var nameCode = 'NAME-' + companyPad4_(nameGroupCounter++);
    var colorOrange = '#fce5cd';

    nameGroup.forEach(function(r) {
      if (r.marked) return;

      markOut[r.idx] = [nameCode];
      reasonOut[r.idx] = ['🟠 SIMILAR_NAME_DIFF_ADDRESS | Nom compatible par mot fort "' + token + '", mais adresse différente. Vérifier avant merge.'];
      bgMark[r.idx] = [colorOrange];
      bgReason[r.idx] = [colorOrange];
      r.marked = true;
    });

    statsOrange++;
  }

  records.forEach(function(r) {
    if (!r.marked && r.companyName && !r.addressGroupKey) {
      markOut[r.idx] = [''];
      reasonOut[r.idx] = [''];
      bgMark[r.idx] = [null];
      bgReason[r.idx] = [null];
    }
  });

  sheet.getRange(2, cMark, n, 1).setValues(markOut);
  sheet.getRange(2, cReas, n, 1).setValues(reasonOut);

  sheet.getRange(2, cMark, n, 1).setBackgrounds(bgMark);
  sheet.getRange(2, cReas, n, 1).setBackgrounds(bgReason);

  return 'Companies Checker terminé. ' +
         'Verts: ' + statsStrong +
         ' | Jaunes: ' + statsReview +
         ' | Oranges: ' + statsOrange +
         ' | Rouges: ' + statsConflict +
         ' | Merge Key Manual non modifié.';
}

function companyParseAdresseNormalisee_(addrNorm) {
  var raw = (addrNorm || '').toString().trim();

  if (!raw) {
    return {
      streetPart: '',
      postal: '',
      city: '',
      province: '',
      country: '',
      addressBaseKey: '',
      addressGroupKey: '',
      addressBaseDisplay: ''
    };
  }

  var parts = raw.split('|').map(function(p) { return (p || '').toString().trim(); });

  var streetPart = parts[0] || raw;
  var postal = parts[1] || '';
  var city = parts[2] || '';
  var province = parts[3] || '';
  var country = parts[4] || '';

  var base = companyExtractAddressBase_(streetPart);
  var cityKey = companyNormalizeText_(city);

  var addressGroupKey = '';
  if (base.key && cityKey) {
    addressGroupKey = base.key + '|' + cityKey;
  } else if (base.key) {
    addressGroupKey = base.key;
  }

  return {
    streetPart: streetPart,
    postal: postal,
    city: city,
    province: province,
    country: country,
    addressBaseKey: base.key,
    addressGroupKey: addressGroupKey,
    addressBaseDisplay: base.display
  };
}

function companyExtractAddressBase_(street) {
  var text = companyNormalizeText_(street);

  if (!text) {
    return { key: '', display: '' };
  }

  text = text
    .replace(/\b(canada|quebec|qc|ontario|on)\b/g, ' ')
    .replace(/\b(suite|bureau|bur|local|unit|unite|apt|app|appartement|porte|floor|etage|niveau)\b.*$/g, ' ')
    .replace(/[,;].*$/g, ' ')
    .replace(/-/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();

  text = text
    .replace(/\bboul\.?\b/g, 'boulevard')
    .replace(/\bblvd\.?\b/g, 'boulevard')
    .replace(/\bav\.?\b/g, 'avenue')
    .replace(/\bave\.?\b/g, 'avenue')
    .replace(/\brd\.?\b/g, 'road')
    .replace(/\brte\.?\b/g, 'route')
    .replace(/\bch\.?\b/g, 'chemin')
    .replace(/\s+/g, ' ')
    .trim();

  var tokens = text.split(/\s+/).filter(Boolean);

  if (!tokens.length) {
    return { key: '', display: '' };
  }

  var civicIndex = -1;

  if (tokens.length >= 2 && companyIsNumber_(tokens[0]) && companyIsNumber_(tokens[1])) {
    civicIndex = 1;
  } else {
    for (var i = 0; i < tokens.length; i++) {
      if (companyIsNumber_(tokens[i])) {
        civicIndex = i;
        break;
      }
    }
  }

  if (civicIndex === -1) {
    return {
      key: tokens.join(' '),
      display: companyTitleCase_(tokens.join(' '))
    };
  }

  var civic = tokens[civicIndex];
  var streetTokens = tokens.slice(civicIndex + 1);

  var streetTypeWords = {
    'rue': true,
    'boulevard': true,
    'avenue': true,
    'road': true,
    'chemin': true,
    'route': true,
    'drive': true,
    'dr': true,
    'place': true,
    'pl': true,
    'court': true,
    'ct': true,
    'crescent': true,
    'terrace': true,
    'terrasse': true
  };

  streetTokens = streetTokens.filter(function(tok) {
    return !streetTypeWords[tok];
  });

  var streetName = streetTokens.join(' ').trim();

  if (!streetName) {
    return {
      key: civic,
      display: civic
    };
  }

  return {
    key: civic + '|' + streetName,
    display: civic + ' ' + companyTitleCase_(streetName)
  };
}

function companyStrongNameTokens_(name, city) {
  var stop = {
    'eglise': true,
    'church': true,
    'paroisse': true,
    'parish': true,
    'baptist': true,
    'baptiste': true,
    'evangelique': true,
    'evangelical': true,
    'fellowship': true,
    'ministere': true,
    'ministry': true,
    'ministries': true,
    'centre': true,
    'center': true,
    'communaute': true,
    'community': true,
    'association': true,
    'organisation': true,
    'organization': true,
    'corporation': true,
    'compagnie': true,
    'company': true,
    'inc': true,
    'ltee': true,
    'ltd': true,
    'corp': true,
    'fondation': true,
    'foundation': true,
    'projet': true,
    'mission': true,
    'maison': true,
    'de': true,
    'du': true,
    'des': true,
    'la': true,
    'le': true,
    'les': true,
    'l': true,
    'd': true,
    'the': true,
    'and': true,
    'et': true,
    'saint': true,
    'sainte': true,
    'st': true,
    'ste': true
  };

  companyNormalizeText_(city).split(/\s+/).forEach(function(t) {
    if (t) stop[t] = true;
  });

  var tokens = companyNormalizeText_(name)
    .replace(/[^a-z0-9\s]/g, ' ')
    .split(/\s+/)
    .map(function(t) { return t.trim(); })
    .filter(function(t) {
      return t.length >= 4 && !stop[t];
    });

  var unique = {};
  var out = [];

  tokens.forEach(function(t) {
    if (!unique[t]) {
      unique[t] = true;
      out.push(t);
    }
  });

  return out;
}

function companyGroupHasNameOverlap_(group) {
  var countByToken = {};

  group.forEach(function(r) {
    var seen = {};
    r.nameTokens.forEach(function(t) {
      if (seen[t]) return;
      seen[t] = true;
      countByToken[t] = (countByToken[t] || 0) + 1;
    });
  });

  for (var t in countByToken) {
    if (countByToken[t] >= 2) return true;
  }

  return false;
}

function companyHasSameSourceDifferentIds_(group) {
  var idsByPlatform = {};

  group.forEach(function(r) {
    var platform = companyNormalizeText_(r.platform);
    var rawId = (r.rawId || '').toString().trim();

    if (!platform || !rawId) return;

    if (!idsByPlatform[platform]) idsByPlatform[platform] = {};
    idsByPlatform[platform][rawId] = true;
  });

  for (var p in idsByPlatform) {
    if (Object.keys(idsByPlatform[p]).length > 1) return true;
  }

  return false;
}

function companyUniqueNonBlank_(arr) {
  var map = {};
  var out = [];

  arr.forEach(function(v) {
    var val = (v || '').toString().trim();
    if (!val) return;
    if (!map[val]) {
      map[val] = true;
      out.push(val);
    }
  });

  return out.sort();
}

function companyNormalizeText_(value) {
  return (value || '')
    .toString()
    .trim()
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[’']/g, ' ')
    .replace(/[^a-z0-9\s]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

function companyNormalizePostal_(value) {
  return (value || '')
    .toString()
    .trim()
    .toUpperCase()
    .replace(/\s+/g, '');
}

function companyIsNumber_(value) {
  return /^\d+$/.test((value || '').toString());
}

function companyPad4_(num) {
  return String(num).padStart(4, '0');
}

function companyTitleCase_(text) {
  return (text || '')
    .toString()
    .split(/\s+/)
    .filter(Boolean)
    .map(function(w) {
      return w.charAt(0).toUpperCase() + w.slice(1);
    })
    .join(' ');
}
